#include <TM1638.h>

const char* ssid="고동재";
const char* password="12341234";

ESP8266WebServer server(80);//1. 서버 객체(인스턴스) 생성

void setup() {
 Serial.begin(9600);
 WiFi.softAP(ssid,password);
 Serial.println();
 Serial.print("AP IP Address");
 Serial.println(WiFi.softAPIP());

 server.on("/",handleRoot);//2. 핸들러 등록
 server.begin();//3.웹 서버 시작

}

void loop() {
  server.handleClient();//4. 주기적인 웹 서버 기능 수행
}

void handleRoot()
{
  if(server.hasArg("LED")){
    String led = server.arg("LED");

    int ledvalue = led.toInt();

    aaa.setLEDs(ledvalue);
    aaa.setDisplayToDecNumber(ledvalue, 0);


  }
  server.send(200,"text/html",MAIN_page);
}
